% The title settings are based on this amazing TeX SE answer:
% https://tex.stackexchange.com/questions/599059/fancy-chapter-style-custom
% Also an idea: https://tex.stackexchange.com/questions/254318/chapter-style-with-tcolorbox

\usepackage[most]{tcolorbox}
\usepackage[explicit]{titlesec}

% Chapter title colors
\colorlet{chaptercol1}{xred}
\colorlet{chaptercol2}{xblue}
\colorlet{chaptercol3}{xgreen}
\colorlet{chaptercol4}{xpurple}
\colorlet{chaptercol5}{xorange}

\definecolor{titlebgdark}{HTML}{48505e}
\definecolor{titlebglight}{HTML}{596774}

\titleformat{\chapter}[display]
{}
{}
{60mm}
{%
	\begin{tcolorbox}[
			enhanced,
			halign=flush left,
			frame style={left color=titlebglight, right color=white},
			interior engine=path,
			interior style={left color=titlebgdark, right color=white},
			boxrule=5.0mm,
			arc=0pt,
			outer arc=0pt,
			leftrule=0pt,
			rightrule=0pt,
			fontupper=\color{white}\chaptertitleseries\bfseries\fontsize{40}{40}\selectfont,
			enlarge left by=-1in-\hoffset-\oddsidemargin,
			enlarge right by=-\paperwidth+25mm+\hoffset+\oddsidemargin+\textwidth,
			width=\paperwidth,
			left=20mm,
			right=\paperwidth-25mm-\hoffset-\oddsidemargin-\textwidth,
			top=15mm,
			bottom=5mm,
			overlay={
					\node[
						fill=titlebgdark,
						draw=titlebglight,
						line width=2.5mm,
						inner sep=0pt,
						text width=32mm,
						minimum height=32mm,
						align=center,
						anchor=west,
						font=\color{white}\chapternumberseries\bfseries\fontsize{100}{106}\selectfont,
					]
					(chapname)
					at ([xshift=20mm]frame.north west)
					{\thechapter};
					\node[font=\chapternumberseries\bfseries\fontsize{20}{20}\selectfont,anchor=south,yshift=5pt, inner sep=2pt] at (chapname.north)
					{\MakeUppercase\chaptertitlename};  % if using amsbook, this should be \chaptername
					\node[anchor=south, yshift=-2mm] (tapir) at (frame.north) {\includesvg{graphics/chapters/chapter\thechapter}};
				}
		]
		\uppercase{#1}
	\end{tcolorbox}%
	% Add chapter description under title - using the \description macro
	% defined in the main chapters loop (main.tex)
	\vspace{3em}
	\chapterdescription
}

\titleformat{name=\chapter,numberless}[display]
{\normalfont\huge\bfseries}
{}
{20pt}
{%
	\begin{tcolorbox}[
			enhanced,
			colback=titlebgdark,
			boxrule=0.25cm,
			colframe=titlebglight,
			arc=0pt,
			outer arc=0pt,
			remember as=title,
			leftrule=0pt,
			rightrule=0pt,
			fontupper=\color{white}\sffamily\bfseries\huge,
			enlarge left by=-1in-\hoffset-\oddsidemargin,
			enlarge right by=-\paperwidth+1in+\hoffset+\oddsidemargin+\textwidth,
			width=\paperwidth,
			left=1in+\hoffset+\oddsidemargin,
			right=\paperwidth-1in-\hoffset-\oddsidemargin-\textwidth,
			top=0.6cm,
			bottom=0.6cm,
		]
		#1
	\end{tcolorbox}%
}
\titlespacing*{\chapter}
{0pt}{0pt}{40pt}
\makeatother
\titleformat{\section}[display]
{\sffamily\large}
{}
{0pt}
{\hrule\vspace*{0.25ex}\parbox[b]{\dimexpr\textwidth\relax}{\textcolor{darkgray}{\thesection}\quad\raggedright\bfseries\MakeUppercase{#1}}}
[\hrule]

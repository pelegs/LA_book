% The title settings are based on this amazing TeX SE answer:
% https://tex.stackexchange.com/questions/599059/fancy-chapter-style-custom
% Also an idea: https://tex.stackexchange.com/questions/254318/chapter-style-with-tcolorbox

\usepackage[most]{tcolorbox}
\usepackage[explicit]{titlesec}

% Chapter title colors
\definecolor{chaptercoldark1}{HTML}{5e5156}
\definecolor{chaptercoldark2}{HTML}{48505e}
\definecolor{chaptercoldark3}{HTML}{485e59}
\definecolor{chaptercoldark4}{HTML}{5e485e}
\definecolor{chaptercoldark5}{HTML}{5e4b48}

\colorlet{chaptercollight1}{chaptercoldark1!50!white}
\colorlet{chaptercollight2}{chaptercoldark2!50!white}
\colorlet{chaptercollight3}{chaptercoldark3!50!white}
\colorlet{chaptercollight4}{chaptercoldark4!50!white}
\colorlet{chaptercollight5}{chaptercoldark5!50!white}

\titleformat{\chapter}[display]
{}
{}
{60mm}
{%
	\begin{tcolorbox}[
			enhanced,
			halign=flush left,
			frame style={left color=chaptercollight\thechapter, right color=white},
			interior engine=path,
			interior style={left color=chaptercoldark\thechapter, right color=white},
			boxrule=5.0mm,
			arc=0pt,
			outer arc=0pt,
			leftrule=0pt,
			rightrule=0pt,
			fontupper=\color{white}\chaptertitleseries\bfseries\fontsize{40}{40}\selectfont,
			enlarge left by=-1in-\hoffset-\oddsidemargin,
			enlarge right by=-\paperwidth+25mm+\hoffset+\oddsidemargin+\textwidth,
			width=\paperwidth,
			left=20mm,
			right=\paperwidth-25mm-\hoffset-\oddsidemargin-\textwidth,
			top=15mm,
			bottom=5mm,
			overlay={
					\node[
						fill=chaptercoldark\thechapter,
						draw=chaptercollight\thechapter,
						line width=2.5mm,
						inner sep=0pt,
						text width=32mm,
						minimum height=32mm,
						align=center,
						anchor=west,
						font=\color{white}\chapternumberseries\bfseries\fontsize{100}{106}\selectfont,
					]
					(chapname)
					at ([xshift=20mm]frame.north west)
					{\thechapter};
					\node[font=\chapternumberseries\bfseries\fontsize{20}{20}\selectfont,anchor=south,yshift=5pt, inner sep=2pt] at (chapname.north)
					{\MakeUppercase\chaptertitlename};  % if using amsbook, this should be \chaptername
					\node[anchor=south, yshift=-2mm] (tapir) at (frame.north) {\includesvg{graphics/chapters/chapter\thechapter}};
				}
		]
		\uppercase{#1}
	\end{tcolorbox}%
	% Add chapter description under title - using the \description macro
	% defined in the main chapters loop (main.tex)
	\vspace{3em}
	\chapterdescription
}

\titleformat{\section}[display]
{\sffamily\large}
{}
{0pt}
{\hrule\vspace*{0.25ex}\parbox[b]{\dimexpr\textwidth\relax}{\thesection\quad\raggedright\bfseries\MakeUppercase{#1}}}
[\hrule]

% Numberless headings (for preface chapter)
\titleformat{name=\chapter, numberless}
{\normalfont\huge\bfseries}
{}
{0pt}
{#1}

\titleformat{name=\section, numberless}{\bfseries}{#1}{0em}{}
